# Every file that carries the distributed version literal, one per line:
#   path|prefix|suffix|scope
# The literal is `prefix + VERSION + suffix`. Scope `line` means the literal
# must start at a line boundary; `any` matches it anywhere on a line.
# scripts/audit-versions.sh checks these and scripts/bump-version.sh rewrites
# them; Cargo.lock is handled separately by both.
Cargo.toml|version = "|"|line
extension/manifest.json|"version": "|"|any
extension/lib.js|export const VERSION = "|";|line
.github/workflows/deploy.yml|(for example, |)|any
.github/workflows/publish.yml|(for example, |)|any
docs/INSTALL_MACOS.md|--version ||any
docs/INSTALL_WINDOWS.md|-Version ||any
